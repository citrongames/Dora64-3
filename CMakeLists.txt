cmake_minimum_required(VERSION 3.20)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" DORA3_VERSION LIMIT_COUNT 1)
project(Dora64Three VERSION ${DORA3_VERSION} LANGUAGES C CXX)

option(DORA3_BUILD_PROBES "Build the historical research probes instead of the game" OFF)
if(DORA3_BUILD_PROBES)
    include(cmake/ResearchProbes.cmake)
else()
    set(DORA3_LIVE_RT64 ON CACHE BOOL "Build the RT64 game frontend")
    set(DORA3_DIAGNOSTICS OFF CACHE BOOL "Enable diagnostic journals")
    set(DORA3_GENERATED_DIR "${CMAKE_CURRENT_SOURCE_DIR}/build-local/startup" CACHE PATH "Generated CPU sources")
    set(DORA3_AUDIO_GENERATED_DIR "${CMAKE_CURRENT_SOURCE_DIR}/build-local/audio-rsp" CACHE PATH "Generated audio sources")
    add_subdirectory(research/runtime)
    if(NOT ANDROID)
        set_target_properties(dora3_startup PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
    endif()
endif()
