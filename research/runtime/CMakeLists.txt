cmake_minimum_required(VERSION 3.20)
project(Dora3StartupRuntime LANGUAGES C CXX)
set(CMAKE_C_STANDARD 17)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(DORA3_GENERATED_DIR "" CACHE PATH "generate_startup.py output")
if(NOT EXISTS "${DORA3_GENERATED_DIR}/generation-receipt.json")
    message(FATAL_ERROR "Run generate_startup.py and set DORA3_GENERATED_DIR")
endif()
file(READ "${DORA3_GENERATED_DIR}/generation-receipt.json" receipt)
foreach(group inputs generated)
    if(group STREQUAL "inputs")
        set(base "${ROOT}")
    else()
        set(base "${DORA3_GENERATED_DIR}")
    endif()
    string(JSON count LENGTH "${receipt}" ${group})
    math(EXPR last "${count}-1")
    foreach(i RANGE 0 ${last})
        string(JSON name MEMBER "${receipt}" ${group} ${i})
        string(JSON expected GET "${receipt}" ${group} "${name}")
        file(SHA256 "${base}/${name}" actual)
        if(NOT actual STREQUAL expected)
            message(FATAL_ERROR "Stale or modified ${base}/${name}; regenerate")
        endif()
        set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${base}/${name}")
    endforeach()
endforeach()
set(RUNTIME "${ROOT}/build-deps/lib/N64ModernRuntime")
add_compile_definitions(HLSL_CPU)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64" AND NOT MSVC)
    add_compile_options(-mssse3 -msse4.1)
endif()
add_subdirectory("${RUNTIME}" runtime)
file(GLOB generated CONFIGURE_DEPENDS "${DORA3_GENERATED_DIR}/RecompiledFuncs/*.c")
foreach(source IN LISTS generated)
    file(RELATIVE_PATH relative "${DORA3_GENERATED_DIR}" "${source}")
    string(JSON expected ERROR_VARIABLE receipt_error GET "${receipt}" generated "${relative}")
    if(receipt_error)
        message(FATAL_ERROR "Unexpected generated source ${relative}; regenerate")
    endif()
endforeach()
add_executable(dora3_startup "${ROOT}/src/startup_runtime.cpp" "${DORA3_GENERATED_DIR}/frontier.c" ${generated})
target_include_directories(dora3_startup PRIVATE "${DORA3_GENERATED_DIR}" "${DORA3_GENERATED_DIR}/RecompiledFuncs" "${RUNTIME}/N64Recomp/include")
target_link_libraries(dora3_startup PRIVATE librecomp ultramodern)
if(NOT MSVC)
    target_compile_options(dora3_startup PRIVATE -fno-strict-aliasing)
endif()
