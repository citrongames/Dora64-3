    message(FATAL_ERROR "Generate CPU/audio RSP sources and set both DORA3_*_GENERATED_DIR paths")
endif()
file(READ "${generated_dir}/generation-receipt.json" receipt)
foreach(group inputs generated)
    if(group STREQUAL "inputs")
        set(base "${ROOT}")
    else()
        set(base "${generated_dir}")
    endif()
    string(JSON count LENGTH "${receipt}" ${group})
    math(EXPR last "${count}-1")
    foreach(i RANGE 0 ${last})
        string(JSON name MEMBER "${receipt}" ${group} ${i})
        string(JSON expected GET "${receipt}" ${group} "${name}")
        file(SHA256 "${base}/${name}" actual)
        if(NOT actual STREQUAL expected)
            message(FATAL_ERROR "Stale or modified ${base}/${name}; regenerate")
        endif()
        set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${base}/${name}")
    endforeach()
endforeach()
endforeach()
file(READ "${DORA3_GENERATED_DIR}/generation-receipt.json" receipt)

foreach(source IN LISTS generated)
    file(RELATIVE_PATH relative "${DORA3_GENERATED_DIR}" "${source}")
    string(JSON expected ERROR_VARIABLE receipt_error GET "${receipt}" generated "${relative}")
    if(receipt_error)
        message(FATAL_ERROR "Unexpected generated source ${relative}; regenerate")
    endif()
endforeach()

option(DORA3_LIVE_RT64 "Experimental desktop live RT64 startup" OFF)
if(DORA3_LIVE_RT64)

        message(FATAL_ERROR "Live research frontend currently supports Linux/Windows only")

    if(ANDROID)
        set(SDL_SHARED ON CACHE BOOL "" FORCE)
        set(SDL_STATIC OFF CACHE BOOL "" FORCE)

        add_subdirectory("${ROOT}/build-deps/lib/SDL2" SDL2)
    elseif(WIN32)
        set(SDL2_ROOT "${RT64_ROOT}/src/contrib/mupen64plus-win32-deps/SDL2-2.26.3")
        add_library(SDL2::SDL2 SHARED IMPORTED)
        set_target_properties(SDL2::SDL2 PROPERTIES
            IMPORTED_IMPLIB "${SDL2_ROOT}/lib/x64/SDL2.lib"
            IMPORTED_LOCATION "${SDL2_ROOT}/lib/x64/SDL2.dll"
            INTERFACE_INCLUDE_DIRECTORIES "${SDL2_ROOT}/include")
        target_compile_definitions(dora3_startup PRIVATE NOMINMAX SDL_MAIN_HANDLED)
        target_link_libraries(dora3_startup PRIVATE psapi)
        set_target_properties(dora3_startup PROPERTIES OUTPUT_NAME "Dora64-3")
        add_custom_command(TARGET dora3_startup POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:SDL2::SDL2>" "$<TARGET_FILE_DIR:dora3_startup>"
            COMMAND ${CMAKE_COMMAND} -E copy_if_different "${RT64_ROOT}/src/contrib/dxc/bin/x64/dxcompiler.dll" "$<TARGET_FILE_DIR:dora3_startup>"
            COMMAND ${CMAKE_COMMAND} -E copy_if_different "${RT64_ROOT}/src/contrib/dxc/bin/x64/dxil.dll" "$<TARGET_FILE_DIR:dora3_startup>")
    else()
        find_package(SDL2 REQUIRED)
    endif()
    if(NOT TARGET rt64)
        set(RT64_STATIC TRUE)
        set(RT64_SDL_WINDOW_VULKAN TRUE)
        # Adapt raster source at configure time; dependency checkout/pins stay untouched.
        set(sprite_rt64 "${CMAKE_CURRENT_BINARY_DIR}/rt64-sprite-edges")
        execute_process(COMMAND "${Python3_EXECUTABLE}" "${ROOT}/tools/prepare_rt64_sprite_edges.py"
            "${sprite_rt64}" COMMAND_ERROR_IS_FATAL ANY)
        set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
            "${ROOT}/tools/prepare_rt64_sprite_edges.py"
            "${ROOT}/src/shaders/world_sprite_edges.hlsli"
            "${RT64_ROOT}/CMakeLists.txt"
            "${RT64_ROOT}/src/shaders/RasterPS.hlsl"
            "${RT64_ROOT}/src/shaders/TextureSampler.hlsli"
            "${RT64_ROOT}/src/shaders/FbWriteColorCS.hlsl"
            "${RT64_ROOT}/src/shaders/FbWriteDepthCS.hlsl"
            "${RT64_ROOT}/src/shaders/FbWritePackedCS.hlsl")
        add_subdirectory("${sprite_rt64}" rt64)

    # Only Dora64-3 uses this build-local queue copy; pinned libraries and Dora64 stay untouched.
    set(adapted_rt64 "${CMAKE_CURRENT_BINARY_DIR}/rt64-framerate")
    execute_process(COMMAND "${Python3_EXECUTABLE}" "${ROOT}/tools/prepare_rt64_framerate.py"
        "${adapted_rt64}" COMMAND_ERROR_IS_FATAL ANY)
    get_target_property(rt64_sources rt64 SOURCES)
    foreach(queue workload present)
        set(original_queue "${RT64_ROOT}/src/hle/rt64_${queue}_queue.cpp")
        list(FIND rt64_sources "${original_queue}" queue_index)
        if(queue_index EQUAL -1)
            message(FATAL_ERROR "RT64 ${queue} source layout changed")
        endif()
        list(REMOVE_ITEM rt64_sources "${original_queue}")
        list(APPEND rt64_sources "${adapted_rt64}/rt64_${queue}_queue.cpp")
        set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${original_queue}")
    endforeach()
    set(original_state "${RT64_ROOT}/src/hle/rt64_state.cpp")
    list(FIND rt64_sources "${original_state}" state_index)

    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${original_state}")

    if(native_target_index EQUAL -1)

    list(REMOVE_ITEM rt64_sources "${original_native_target}")
    list(APPEND rt64_sources "${adapted_rt64}/rt64_native_target.cpp")

    endif()

        list(APPEND rt64_sources "${adapted_rt64}/rt64_${detail}.cpp")
        set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${original_detail}")

    foreach(detail framebuffer_renderer projection_processor)

        list(REMOVE_ITEM rt64_sources "${original_detail}")
        list(APPEND rt64_sources "${adapted_rt64}/rt64_${detail}.cpp")

    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${ROOT}/tools/rt64_widescreen.py")

    endif()

    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS

    target_sources(dora3_startup PRIVATE "${ROOT}/src/framebuffer_readback.cpp" "${ROOT}/src/live_graphics.cpp" "${ROOT}/src/live_io.cpp" "${ROOT}/src/port_settings.cpp" "${ROOT}/src/port_input.cpp" "${ROOT}/src/port_menu.cpp" "${ROOT}/src/port_display.cpp" "${ROOT}/src/port_touch.cpp" "${ROOT}/src/android_platform.cpp")
