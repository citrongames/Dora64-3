cmake_minimum_required(VERSION 3.20)
project(Dora3StartupRuntime LANGUAGES C CXX)
set(CMAKE_C_STANDARD 17)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(DORA3_GENERATED_DIR "" CACHE PATH "generate_startup.py output")
set(DORA3_AUDIO_GENERATED_DIR "" CACHE PATH "generate_audio_rsp.py output")
foreach(generated_dir "${DORA3_GENERATED_DIR}" "${DORA3_AUDIO_GENERATED_DIR}")
if(NOT EXISTS "${generated_dir}/generation-receipt.json")
    message(FATAL_ERROR "Generate CPU/audio RSP sources and set both DORA3_*_GENERATED_DIR paths")
endif()
file(READ "${generated_dir}/generation-receipt.json" receipt)
foreach(group inputs generated)
    if(group STREQUAL "inputs")
        set(base "${ROOT}")
    else()
        set(base "${generated_dir}")
    endif()
    string(JSON count LENGTH "${receipt}" ${group})
    math(EXPR last "${count}-1")
    foreach(i RANGE 0 ${last})
        string(JSON name MEMBER "${receipt}" ${group} ${i})
        string(JSON expected GET "${receipt}" ${group} "${name}")
        file(SHA256 "${base}/${name}" actual)
        if(NOT actual STREQUAL expected)
            message(FATAL_ERROR "Stale or modified ${base}/${name}; regenerate")
        endif()
        set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${base}/${name}")
    endforeach()
endforeach()
endforeach()
file(READ "${DORA3_GENERATED_DIR}/generation-receipt.json" receipt)
set(RUNTIME "${ROOT}/build-deps/lib/N64ModernRuntime")
add_compile_definitions(HLSL_CPU)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64" AND NOT MSVC)
    add_compile_options(-mssse3 -msse4.1)
endif()
add_subdirectory("${RUNTIME}" runtime)
file(GLOB generated CONFIGURE_DEPENDS "${DORA3_GENERATED_DIR}/RecompiledFuncs/*.c")
foreach(source IN LISTS generated)
    file(RELATIVE_PATH relative "${DORA3_GENERATED_DIR}" "${source}")
    string(JSON expected ERROR_VARIABLE receipt_error GET "${receipt}" generated "${relative}")
    if(receipt_error)
        message(FATAL_ERROR "Unexpected generated source ${relative}; regenerate")
    endif()
endforeach()
add_executable(dora3_startup "${ROOT}/src/startup_runtime.cpp" "${DORA3_AUDIO_GENERATED_DIR}/dora3_aspMain.cpp" "${DORA3_GENERATED_DIR}/frontier.c" ${generated})
target_include_directories(dora3_startup PRIVATE "${DORA3_GENERATED_DIR}" "${DORA3_GENERATED_DIR}/RecompiledFuncs" "${RUNTIME}/N64Recomp/include")
target_link_libraries(dora3_startup PRIVATE librecomp ultramodern)
if(NOT MSVC)
    target_compile_options(dora3_startup PRIVATE -fno-strict-aliasing)
endif()

find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(adapted_runtime "${CMAKE_CURRENT_BINARY_DIR}/hardware-runtime")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${ROOT}/tools/prepare_runtime_timing.py"
    "${adapted_runtime}" COMMAND_ERROR_IS_FATAL ANY)
get_target_property(runtime_sources ultramodern SOURCES)
foreach(name events scheduling mesgqueue)
    list(FIND runtime_sources "${RUNTIME}/ultramodern/src/${name}.cpp" source_index)
    if(source_index EQUAL -1)
        message(FATAL_ERROR "Runtime source layout changed: ${name}")
    endif()
    list(REMOVE_ITEM runtime_sources "${RUNTIME}/ultramodern/src/${name}.cpp")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${RUNTIME}/ultramodern/src/${name}.cpp")
endforeach()
set_property(TARGET ultramodern PROPERTY SOURCES "${runtime_sources}")
target_sources(ultramodern PRIVATE "${adapted_runtime}/events-after-parse.cpp" "${adapted_runtime}/scheduling.cpp" "${adapted_runtime}/mesgqueue.cpp")
target_include_directories(ultramodern PRIVATE "${ROOT}/src")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${ROOT}/tools/prepare_runtime_timing.py")
target_sources(dora3_startup PRIVATE "${ROOT}/src/graphics_sync.cpp" "${ROOT}/src/frame_lights.cpp" "${ROOT}/src/n64_hardware.cpp" "${ROOT}/src/n64_hardware_backend.cpp")

option(DORA3_LIVE_RT64 "Experimental desktop live RT64 startup" OFF)
if(DORA3_LIVE_RT64)
    if(NOT (CMAKE_SYSTEM_NAME STREQUAL "Linux" OR WIN32))
        message(FATAL_ERROR "Live research frontend currently supports Linux/Windows only")
    endif()
    set(RT64_ROOT "${ROOT}/build-deps/lib/rt64")
    if(WIN32)
        set(SDL2_ROOT "${RT64_ROOT}/src/contrib/mupen64plus-win32-deps/SDL2-2.26.3")
        add_library(SDL2::SDL2 SHARED IMPORTED)
        set_target_properties(SDL2::SDL2 PROPERTIES
            IMPORTED_IMPLIB "${SDL2_ROOT}/lib/x64/SDL2.lib"
            IMPORTED_LOCATION "${SDL2_ROOT}/lib/x64/SDL2.dll"
            INTERFACE_INCLUDE_DIRECTORIES "${SDL2_ROOT}/include")
        target_compile_definitions(dora3_startup PRIVATE NOMINMAX SDL_MAIN_HANDLED)
        target_link_libraries(dora3_startup PRIVATE psapi)
        set_target_properties(dora3_startup PROPERTIES OUTPUT_NAME "Dora64-3")
        add_custom_command(TARGET dora3_startup POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:SDL2::SDL2>" "$<TARGET_FILE_DIR:dora3_startup>"
            COMMAND ${CMAKE_COMMAND} -E copy_if_different "${RT64_ROOT}/src/contrib/dxc/bin/x64/dxcompiler.dll" "$<TARGET_FILE_DIR:dora3_startup>"
            COMMAND ${CMAKE_COMMAND} -E copy_if_different "${RT64_ROOT}/src/contrib/dxc/bin/x64/dxil.dll" "$<TARGET_FILE_DIR:dora3_startup>")
    else()
        find_package(SDL2 REQUIRED)
    endif()
    if(NOT TARGET rt64)
        set(RT64_STATIC TRUE)
        set(RT64_SDL_WINDOW_VULKAN TRUE)
        add_subdirectory("${RT64_ROOT}" rt64)
    endif()
    target_sources(dora3_startup PRIVATE "${ROOT}/src/live_graphics.cpp" "${ROOT}/src/live_io.cpp")
    target_compile_definitions(dora3_startup PRIVATE DORA3_LIVE_RT64)
    target_include_directories(dora3_startup PRIVATE "${RT64_ROOT}/src" "${RT64_ROOT}/src/contrib"
        "${RT64_ROOT}/src/contrib/hlslpp/include" "${RT64_ROOT}/src/rhi" "${RT64_ROOT}/src/render")
    target_link_libraries(dora3_startup PRIVATE rt64 SDL2::SDL2)
endif()
